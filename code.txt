// Declaracoes globais
int globalInt;
float globalFloat = 1.5;
string globalString = "CORA";

/*
  Comentario de bloco inicial:
  Funcao com retorno e tipo explicito.
  Recebe dois parametros inteiros.
*/
func int soma(int a, int b) {
    return (a + b) * 2; // Expressao com parenteses para precedencia
}

// Funcao com retorno e tipo implicito
func calcularMedia(float n1, float n2) {
    return (n1 + n2) / 2.0;
}

// Funcao sem parametros
func string statusPadrao() {
    return "Aprovado";
}

// Procedimento
proc principal() {
    // Declaracoes locais (devem vir no inicio do escopo)
    int opcao = 1;
    int i;
    float media;
    float nota1 = 7.5;
    float nota2 = 8.5;
    string status;

    // Comandos executaveis
    i = 0;
    media = calcularMedia(nota1, nota2);

    // Condicao (if / else) com operador logico OU (||) e diferenca (!=)
    if (media >= 7.0 || media != 0.0) {
        status = statusPadrao();
    } else {
        status = "Reprovado";
    }

    // Switch com literais inteiros
    switch (opcao) {
        case 1:
            globalInt = soma(10, 20);
            break;
        case 2:
            globalInt = soma(30, 40);
            break;
        default:
            globalInt = 0;
            break;
    }

    // Switch com literais string
    switch (status) {
        case "Aprovado":
            opcao = 10;
            break;
        case "Reprovado":
            opcao = 20;
            break;
        default:
            opcao = 0;
            break;
    }

    // Laco while
    while (!(globalInt == 0)) {
        globalInt = globalInt - 10;
        if (globalInt < 10) {
            break;
        }
    }

    // Laco for com operador modulo (%)
    for (int k = 0; k < 5; k = k + 1) {
        int temp = (k * 2) % 3;
        if (temp == 0) {
            continue;
        }
    }

    // Laco for com inicializacao por atribuicao
    for (i = 0; i < 3; i = i + 1) {
        /* Laco vazio com comentario de bloco interno */
    }

    // Laco for com inicializacao vazia
    for ( ; i > 0; i = i - 1) {
        // Laco vazio
    }

    // Chamada de funcao/procedimento como comando
    soma(1, 2);
}
